cmake_minimum_required(VERSION 3.28)
project(value_direction_hopper LANGUAGES CXX CUDA)
find_package(CUDAToolkit 12.8 REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 90a)
set(TRT_LLM_SOURCE "" CACHE PATH "Pinned Apache-2.0 TensorRT-LLM checkout")
set(CUTLASS_INCLUDE_DIR "" CACHE PATH "CUTLASS include directory with cute/tensor.hpp")
option(VD_FAST_SFU "Explicit SFU exp/log variant; validate separately" OFF)
option(VD_INLINE_ROLES "Inline warp-group roles to avoid device call overhead" ON)
option(VD_BUILD_TENSORRT_PLUGIN "Build the TensorRT V3 integration" ON)
if(NOT EXISTS "${TRT_LLM_SOURCE}/cpp/kernels/fmha_v2/src/fmha/hopper/utils_hgmma_bf16.h")
  message(FATAL_ERROR "Set TRT_LLM_SOURCE to the pinned reference checkout")
endif()
if(NOT EXISTS "${CUTLASS_INCLUDE_DIR}/cute/tensor.hpp")
  message(FATAL_ERROR "Set CUTLASS_INCLUDE_DIR")
endif()
add_library(value_direction_hopper SHARED csrc/value_direction.cu)
target_include_directories(value_direction_hopper PRIVATE
  "${TRT_LLM_SOURCE}/cpp/kernels/fmha_v2/src" "${CUTLASS_INCLUDE_DIR}")
target_link_libraries(value_direction_hopper PRIVATE CUDA::cudart CUDA::cuda_driver)
target_compile_options(value_direction_hopper PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>)
if(VD_FAST_SFU)
  target_compile_definitions(value_direction_hopper PRIVATE VD_FAST_SFU=1)
endif()
if(VD_INLINE_ROLES)
  target_compile_definitions(value_direction_hopper PRIVATE VD_INLINE_ROLES=1)
endif()
set_target_properties(value_direction_hopper PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
if(VD_BUILD_TENSORRT_PLUGIN)
  find_path(TENSORRT_INCLUDE_DIR NvInfer.h HINTS /usr/local/tensorrt/include REQUIRED)
  find_library(TENSORRT_LIBRARY nvinfer HINTS /usr/local/tensorrt/lib REQUIRED)
  add_library(value_direction_trt SHARED csrc/trt_plugin.cpp)
  target_include_directories(value_direction_trt PRIVATE "${TENSORRT_INCLUDE_DIR}")
  target_link_libraries(value_direction_trt PRIVATE value_direction_hopper "${TENSORRT_LIBRARY}" CUDA::cudart)
  set_target_properties(value_direction_trt PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
